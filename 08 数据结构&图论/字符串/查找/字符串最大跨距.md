# [题目: 字符串最大跨距](https://programming.pku.edu.cn/probset/6145fa4759d340339126b3fa75d7da7e/9b8175e20f894853aa1a4e93a0d9cf07/)

有三个字符串S,S1,S2，其中，S长度不超过300,S1和S2的长度不超过10，想检测S1和S2是否同时在S中出现，且S1位于S2的左边，并在S中互不交叉（即，S1的右边界点在S2的左边界点的左侧）。计算满足上述条件的最大跨距（即，最大间隔距离：最右边的S2的起始点与最左边的S1的终止点之间的字符数目）。如果没有满足条件的偶对S1，S2存在，则输出 -1

例如，S = "abcd123ab888efghij45ef67kl", S1="ab", S2="ef"，其中，S1在S中出现了2次，S2也在S中出现了2次，最大跨距则为：18

关于输入
三个串：S, S2, S3，其间以逗号间隔（注意，S, S2, S3中均不含逗号）；

关于输出
S1和S2在S最大跨距；若在S中没有满足条件的S1和S2，则输出-1

例子输入
abcd123ab888efghij45ef67kl,ab,ef
例子输出
18

```cpp
#include <bits/stdc++.h>
using namespace std;

int end_pos_of_first_substr(const string&s, const string& s1){
    int n = s.size();
    int m = s1.size();
    //注意这里的关键取等
    for(int i = 0; i <= n-m; i ++){
        bool flag = true;
        for(int j = i; j < i + m; j ++){
            if(s1[j-i]!=s[j]){
                flag = false;
                break;
            }
        }
        if(flag){
            return i+m-1;
        }
    }

    return -1;

}
int start_pos_of_last_substr(const string&s, const string&s2){
    int n = s.size();
    int m = s2.size();
    int res = -1;
    for(int i = 0; i <= n-m; i ++){
        bool flag = true;
        for(int j = i; j < i + m; j ++){
            if(s2[j-i]!=s[j]){
                flag = false;
                break;
            }
        }
        if(flag){
            res = i;
        }
    }

    return res;

}

int main()
{
    string s = "",s1="",s2="";
    getline(cin, s, ',');
    getline(cin, s1, ',');
    getline(cin, s2);
    int t1 = end_pos_of_first_substr(s,s1);
    int t2 = start_pos_of_last_substr(s,s2);
    if(t1!=-1&&t2!=-1&&t2-t1-1>=0){
        cout<<t2-t1-1;
    }else{
        cout<<-1;
    }
    return 0;
}
```
