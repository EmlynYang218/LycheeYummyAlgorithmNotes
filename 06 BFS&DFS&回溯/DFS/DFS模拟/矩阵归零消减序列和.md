# [题目: 矩阵归零消减序列和](https://programming.pku.edu.cn/probset/5c26f2630c724b359120a8452eb4cbf8/3e2a500583f0470ead7e1499d79fa63a/)

#### 题目描述

    给定一个n*n的矩阵（3<=n<=100，元素的值都是非负整数）。通过n-1次实施下述过程，可把这个矩阵转换成一个1*1的矩阵。每次的过程如下：  
    首先对矩阵进行归零：即对每一行（或一列）上的所有元素，都在其原来值的基础上减去该行（或列）上的最小值，保证相减后的值仍然是非负整数，且这一行（或列）上至少有一个元素的值为0。  
    然后对矩阵进行消减：即把n*n矩阵的第二行和第二列删除（如果二维数组为a[][]，则删除的是a[1][1]所在的行和列），使之转换为一个(n-1)*(n-1)的矩阵。  
    下一次过程，对生成的(n-1)*(n-1)矩阵实施上述过程。显然，经过n-1次上述过程， n*n的矩阵会被转换为一个1*1的矩阵。  
    请求出每次消减前a[1][1]值之和。  

#### 关于输入

第一行是一个整数n。  
其后是n个n*n的矩阵。  
每个矩阵占n行，每行有n个正整数，每个整数间用空格分隔。  

#### 关于输出

输出为n行，每行上的整数为对应矩阵归零消减过程中，每次消减前a[1][1]值之和。

#### 例子输入

3
1 2 3
2 3 4
3 4 5
1 2 3
5 4 2
9 4 5
1 2 3
5 4 2
9 5 4

#### 例子输出

0
2
1

#include <bits/stdc++.h>
using namespace std;

void put_zero_and_delete(vector<vector<int>> &m, int &sum)
{
    int l = m.size();
    // 递归终点：输入的是一个1*1的矩阵
    if (l < 2)
        return;
    // 存储生成的(n-1)*(n-1)矩阵
    vector<vector<int>> after(l - 1, vector<int>(l - 1, -1));
    // 遍历行
    for (int i = 0; i < l; i++)
    {

        // 找该行最小值
        int min_num = INT_MAX;
        for (int j = 0; j < l; j++)
        {
            if (m[i][j] < min_num)
                min_num = m[i][j];
        }

        // 若最小值大于0，处理该行
        if (min_num > 0)
        {
            for (int j = 0; j < l; j++)
            {
                m[i][j] -= min_num;
            }
        }
    }

    // 遍历列
    for (int j = 0; j < l; j++)
    {

        // 找该列最小值
        int min_num = INT_MAX;
        for (int i = 0; i < l; i++)
        {
            if (m[i][j] < min_num)
                min_num = m[i][j];
        }
        // 若最小值大于0，处理该列
        if (min_num > 0)
        {
            for (int i = 0; i < l; i++)
            {
                m[i][j] -= min_num;
            }
        }
    }

    // 更新累加和
    sum += m[1][1];

    // 删除第二行和第二列
    after[0][0] = m[0][0];

    for (int i = 2; i < l; i++)
    {
        after[0][i - 1] = m[0][i];
    }
    for (int j = 2; j < l; j++)
    {
        after[j - 1][0] = m[j][0];
    }
    for (int i = 2; i < l; i++)
    {
        for (int j = 2; j < l; j++)
        {
            after[i - 1][j - 1] = m[i][j];
        }
    }

    // 递归过程
    put_zero_and_delete(after, sum);
}

int main()
{
    int n = 0;
    cin >> n;

    vector<vector<vector<int>>> cases(n, vector<vector<int>>(n, vector<int>(n, -1)));

    for (int i = 0; i < n; i++)
    {
        for (int p = 0; p < n; p++)
        {
            for (int q = 0; q < n; q++)
            {
                cin >> cases[i][p][q];
            }
        }
    }
    for (int i = 0; i < n; i++)
    {
        int sum = 0;
        put_zero_and_delete(cases[i], sum);
        cout << sum << "\n";
    }

    return 0;
}
