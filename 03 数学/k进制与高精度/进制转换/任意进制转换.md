例题(12.9) 数制转换
题目描述
求任意两个长度在1~10000之间的，不同进制非负整数的转换（2进制～36进制）
不同进制的表示符号为（0，1，...，9，a，b，...，z）或者（0，1，...，9，A，B，...，Z）。

关于输入
输入只有一行，包含三个整数a，n，b。a表示其后的n 是a进制整数，b表示欲将a进制整数n转换成b进制整数。
a，b是十进制整数，2 ≤ a，b < 36。

关于输出
输出包含一行，该行有一个整数为转换后的b进制数。输出时字母符号全部用大写表示，即（0，1，...，9，A，B，...，Z）。

例子输入

```
15 Aab3 7
```

例子输出

```
210306
```

```cpp
#include <iostream>
#include <string>
#include <algorithm>
using namespace std;

int digit_to_int(char d){
    if('0'<=d&&d<='9'){
        return d-'0';
    }
    if('A'<=d&&d<='Z'){
        return d-'A'+10;
    }
    if('a'<=d&&d<='z'){
        return d-'a'+10;
    }
    return 0;
}
char int_to_digit(int d){
    if(0<=d&&d<=9){
        return (char)(d+'0');
    }
    if(9<d&&d<36){
        return (char)(d-10+'A');
    }
    return '#';
}

int main() {
    int a, b;
    string n,ans;
    cin >> a>>n>>b;
    while(n!=""){
        //模拟n/b并得到n%b
        string next;//n/b的整数部分
        int r = 0;//上一位余数
        //竖式除法：本位+上一位余数*a / b 等于 本位结果
        for(int i = 0; i < n.size(); i++){
            int x = r*a + digit_to_int(n[i]);
            //去除商的前导0
            if(x/b!=0||next!=""){
                next += int_to_digit(x/b);
            }
            r = x%b;
        }
        ans += int_to_digit(r);
        n = next;
    }
    reverse(ans.begin(),ans.end());
    cout<<ans<<endl;

    return 0;

}
```
