# 题目 元素和最大的子矩阵

#### 题目描述

求一个 m 级矩阵的所有行列连续的子矩阵中元素和最大者。

#### 关于输入

第一行为正整数 m，m ≤ 4。从第二行起按行输入该矩阵。

#### 关于输出

按行输出子矩阵。最后一行输出这个子矩阵的所有元素之和。

#### 例子输入

4
0 -2 -7 0
9 2 -6 2
-4 1 -4 1
-1 8 0 -2

#### 例子输出

9 2
-4 1
-1 8
15

```
#include <bits/stdc++.h>
using namespace std;

static constexpr int MAX_M = 5;

static int matrix[MAX_M][MAX_M];
static int row_acc[MAX_M];
static int m;

int main()
{
    cin.tie(nullptr);
    cout.tie(nullptr);
    ios::sync_with_stdio(false);

    cin >> m;
    for(int i = 0; i < m; i++){
        for(int j = 0; j < m; j++){
            cin >> matrix[i][j];
        }
    }

    int ans = INT_MIN;
    int left = 0, right = 0, up = 0, bottom = 0;//答案四个角坐标

    //二维变一维，再运行Kadane算法求解最大连续子数组和
    //枚举左右边界的位置，外层循环固定左边界，内层循环枚举右边界
    for(int l = 0; l < m; l++){
        //row_acc[i]表示[l,r]之间第i行所有元素之和
        //每次左边界变化将其清零
        for(int i = 0; i < m; i++){
            row_acc[i] = 0;
        }

        for(int r = l; r<m;r++){
            //每次扩展右边界，更新row_acc
            for(int i = 0; i < m; i++){
                row_acc[i]+=matrix[i][r];
            }
            //在row_acc上运行Kadane算法,跟踪上标下标
            int pre = 0;

            int temp_up = 0;
            for(int i = 0; i < m; i++){
                if(row_acc[i]> pre+row_acc[i]){
                    //下一项重新开始一段
                    pre = row_acc[i];
                    temp_up = i;
                }else{
                    //下一项接到上一段末尾
                    pre += row_acc[i];
                }
                if(pre>ans){
                    up = temp_up;
                    ans = pre;
                    left = l;
                    right = r;
                    bottom = i;
                }
            }
        }
    }
    for(int i = up; i <= bottom; i++){
        for(int j = left; j < right; j++){
            cout<< matrix[i][j]<<' ';
        }
        cout<<matrix[i][right]<<'\n';
    }
    cout<<ans;
    return 0;
}
```
