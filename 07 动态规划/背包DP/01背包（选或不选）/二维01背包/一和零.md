[474. 一和零](https://leetcode.cn/problems/ones-and-zeroes/)

给你一个二进制字符串数组 `strs` 和两个整数 `m` 和 `n` 。

请你找出并返回 `strs` 的最大子集的长度，该子集中 **最多** 有 `m` 个 `0` 和 `n` 个 `1` 。

如果 `x` 的所有元素也是 `y` 的元素，集合 `x` 是集合 `y` 的 **子集** 。

**示例 1：**

**输入：**strs = ["10", "0001", "111001", "1", "0"], m = 5, n = 3
**输出：**4
**解释：**最多有 5 个 0 和 3 个 1 的最大子集是 {"10","0001","1","0"} ，因此答案是 4 。
其他满足题意但较小的子集包括 {"0001","1"} 和 {"10","1","0"} 。{"111001"} 不满足题意，因为它含 4 个 1 ，大于 n 的值 3 。

**示例 2：**

**输入：**strs = ["10", "0", "1"], m = 1, n = 1
**输出：**2
**解释：**最大的子集是 {"0", "1"} ，所以答案是 2 。

**提示：**

- `1 <= strs.length <= 600`
- `1 <= strs[i].length <= 100`
- `strs[i]` 仅由 `'0'` 和 `'1'` 组成
- `1 <= m, n <= 100`

```cpp
static constexpr int MAX_M = 104;
static constexpr int MAX_N = 104;
static constexpr int MAX_L = 604;

static int dp[MAX_L][MAX_M][MAX_N];
static int p[MAX_L];
static int q[MAX_L];

class Solution {
public:
    int findMaxForm(vector<string>& strs, int m, int n) {
        /*
        0-1背包增加维度
        状态设计：
            dp[i][a][b] 表示 前i个数，最多a个0，b个1的最大子集的长度
            答案：dp[strs.size()][m][n]
        状态转移：
            设第i个数有p[i]个0，q[i]个1
            如果选择第i个数，dp[i-1][a-p[i]][b-q[i]] + 1，如果不选dp[i-1][a][b]
            dp[i][a][b] = max(dp[i-1][a-p[i]][b-q[i]] + 1，dp[i-1][a][b])
        边界处理：
            dp[0][a][b] = 0 最大子集是空集
            dp[i][0][0] = 0 空集，
            对于dp[i][0][b] 作递推
            dp[i][0][b] = max(dp[i-1][0][b-q[i]]+1,dp[i-1][0][b])
            对于dp[i][a][0] 作递推
            dp[i][a][0] = max(dp[i-1][a-p[i]][0]+1,dp[i-1][a][0])

            注意！从0开始递推即可自动覆盖边界
        */

        int l = strs.size();
        memset(p, 0, sizeof(p));
        memset(q, 0, sizeof(q));
        for (int i = 0; i < l; i++) {
            for (int j = 0; j < strs[i].size(); j++) {
                if (strs[i][j] == '1') {
                    q[i + 1]++;
                } else {
                    p[i + 1]++;
                }
            }
        }

        for (int i = 0; i <= l; i++) {
            for (int a = 0; a <= m; a++) {
                for (int b = 0; b <= n; b++) {
                    if(i==0){
                        continue;
                    }
                    dp[i][a][b] = dp[i - 1][a][b];
                    //这里必须检查q[i]==0，否则会把0选进去
                    if (a >= p[i] && b >= q[i]) {
                        dp[i][a][b] = max(dp[i - 1][a - p[i]][b - q[i]] + 1,
                                          dp[i - 1][a][b]);
                    }
                }
            }
        }


        return dp[l][m][n];
    }
};
```
